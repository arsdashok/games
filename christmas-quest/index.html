    // --- STATE ---

    let currentLevel = null;

    // --- AUDIO ---

    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    

    function playBellSound() {

        if (audioCtx.state === 'suspended') audioCtx.resume();

        const oscillator = audioCtx.createOscillator();

        const gainNode = audioCtx.createGain();

        

        oscillator.type = 'sine';

        oscillator.frequency.setValueAtTime(800, audioCtx.currentTime); // High pitch

        oscillator.frequency.exponentialRampToValueAtTime(400, audioCtx.currentTime + 1);

        

        gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);

        gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 1);

        

        oscillator.connect(gainNode);

        gainNode.connect(audioCtx.destination);

        

        oscillator.start();

        oscillator.stop(audioCtx.currentTime + 1);

    }


    // --- LOGIC ---


    function showScreen(id) {

        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));

        document.getElementById(id).classList.add('active');

    }


    function startGame(level) {

        currentLevel = level;

        studyIndex = 0;

        // Copy questions to queue to allow modifying order

        questionQueue = [...data[level].questions]; 

        maxScore = questionQueue.length;

        score = 0;

        

        renderFlashcard();

        showScreen('study-screen');

    }


    // Study Phase

    function renderFlashcard() {

        const list = data[currentLevel].study;

        if (studyIndex >= list.length) {

            startQuiz();

            return;

        }

        const item = list[studyIndex];

        const html = `

            <span class="emoji-lg">${item.emoji}</span>

            <span class="word-en">${item.en}</span>

            <span class="word-ru">${item.ru}</span>

            <div class="context">"${item.ex}"</div>

        `;

        document.getElementById('flashcard-container').innerHTML = html;

        

        const btnText = (studyIndex === list.length - 1) ? "Play Game!" : "Next Word";

        document.getElementById('study-next-btn').innerText = btnText;

    }


    function nextFlashcard() {

        studyIndex++;

        renderFlashcard();

    }


    // Game Phase

    function startQuiz() {

        showScreen('game-screen');

        loadNextQuestion();

    }


    function loadNextQuestion() {

        if (questionQueue.length === 0) {

            // Victory!

            document.getElementById('star-top').style.display = 'block';

            setTimeout(() => showScreen('win-screen'), 1500);

            return;

        }


        currentQuestion = questionQueue.shift(); // Take first

        selectedOption = null;

        

        document.getElementById('q-text').innerText = currentQuestion.q;

        document.getElementById('feedback').innerText = "";

        document.getElementById('check-btn').style.display = 'block';

        document.getElementById('check-btn').innerText = "Check";

        document.getElementById('check-btn').disabled = true;

        document.getElementById('check-btn').onclick = checkAnswer;


        const container = document.getElementById('options-container');

        container.innerHTML = '';


        currentQuestion.options.forEach((opt, index) => {

            const btn = document.createElement('button');

            btn.className = 'btn btn-outline option-btn';

            btn.innerText = opt;

            btn.onclick = () => selectOption(index, btn);

            container.appendChild(btn);

        });

    }


    function selectOption(index, btnElement) {

        // Remove selected class from all

        document.querySelectorAll('.option-btn').forEach(b => b.classList.remove('selected'));

        btnElement.classList.add('selected');

        selectedOption = index;

        document.getElementById('check-btn').disabled = false;

    }


    function checkAnswer() {

        const optionsBtns = document.querySelectorAll('.option-btn');

        const feedback = document.getElementById('feedback');

        const checkBtn = document.getElementById('check-btn');


        if (selectedOption === currentQuestion.a) {

            feedback.innerText = "Oops! Try to remember this one.";

            feedback.className = "feedback-msg feedback-wrong";

            // Push back to end of queue to repeat later

            questionQueue.push(currentQuestion);

            

            checkBtn.innerText = "Got it";

            checkBtn.onclick = loadNextQuestion;

        }

    }


    function addOrnament() {

        const tree = document.getElementById('game-tree');

        const ornaments = ["🔴", "🔵", "🟡", "🟣", "🍬", "🍭", "🔔"];

        const randomOrn = ornaments[Math.floor(Math.random() * ornaments.length)];

        

        const el = document.createElement('div');

        el.className = 'ornament';

        el.innerText = randomOrn;

        

        // Random position within the triangle (rough approx)

        // Tree is 300px tall. Width varies.

        // At bottom (y=0) width is +/- 120. At top (y=300) width is 0.

        

        const y = Math.random() * 260 + 20; // 20 to 280

        // Calculate max X at this Y

        // Slope factor: at y=0, x=120. at y=300, x=0. 

        const maxX = 100 * (1 - (y / 300)); 

        const x = (Math.random() * maxX * 2) - maxX; // -maxX to +maxX


        el.style.bottom = y + 'px';

        el.style.left = `calc(50% + ${x}px)`; // Center + offset

        

        tree.appendChild(el);

    }


    function updateProgress() {

        score++;

        const pct = Math.min(100, (score / maxScore) * 100);

        document.getElementById('progress-fill').style.width = pct + '%';

    }

        container.appendChild(flake);

        

        setTimeout(() => {

            flake.remove();

        }, 5000);

    }
